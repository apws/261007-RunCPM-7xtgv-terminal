defproc greet_users(names$, count)
  local i, label$, total

  rem Mixworx: a small greeting loop
  total = 0
  for i = 1 to count
    label$ = "Hello, " & names$(i)
    if names$(i) <> "" then
      print label$
      total = total + 1
    else
      print "Guest #"; i
    endif
  endfor i

  rem Report how many names were greeted
  print "Greeted: "; total
endproc greet_users
